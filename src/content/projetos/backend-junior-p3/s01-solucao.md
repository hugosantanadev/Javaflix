package br.com.ipe.carteira;

import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

@RestControllerAdvice
public class TratadorDeErros {

    @ExceptionHandler(NaoEncontradoException.class)
    public ProblemDetail naoEncontrado(NaoEncontradoException e) {
        return ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, e.getMessage());
    }
}

**`Usuario.java`**

```java
package br.com.ipe.carteira;

import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import jakarta.persistence.Version;
import java.math.BigDecimal;

@Entity
@Table(name = "usuarios")
public class Usuario {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    private String nome;
    private String documento;
    private String email;

    @Enumerated(EnumType.STRING) //> Grava "COMUM", e não a posição 0.
    private TipoUsuario tipo;

    private BigDecimal saldo;

    @Version
    private long versao;

    protected Usuario() { //> Só para o JPA: ninguém cria usuário vazio.
    }

    public Long getId() {
        return id;
    }

    public String getNome() {
        return nome;
    }

    public TipoUsuario getTipo() {
        return tipo;
    }

    public BigDecimal getSaldo() {
        return saldo;
    }
}
```

**`UsuarioResposta.java`**

```java
package br.com.ipe.carteira;

import java.math.BigDecimal;

public record UsuarioResposta(Long id, String nome, TipoUsuario tipo, BigDecimal saldo) {

    static UsuarioResposta de(Usuario u) { //> Documento e e-mail ficam de fora da resposta.
        return new UsuarioResposta(u.getId(), u.getNome(), u.getTipo(), u.getSaldo());
    }
}
```

**`UsuarioService.java`**

```java
package br.com.ipe.carteira;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class UsuarioService {

    private final UsuarioRepository usuarios;

    public UsuarioService(UsuarioRepository usuarios) {
        this.usuarios = usuarios;
    }

    @Transactional(readOnly = true) //> Leitura: o Hibernate dispensa a verificação de sujeira.
    public UsuarioResposta buscar(Long id) {
        return usuarios.findById(id)
                .map(UsuarioResposta::de)
                .orElseThrow(() -> new NaoEncontradoException("usuário " + id + " não existe"));
    }
}
```

A sessão real:

```text
> GET /usuarios/1
< 200 {"id":1,"nome":"Ana Lima","tipo":"COMUM","saldo":500.00}
> GET /usuarios/3
< 200 {"id":3,"nome":"Loja Ipê","tipo":"LOJISTA","saldo":0.00}
> GET /usuarios/99
< 404 {"detail":"usuário 99 não existe","instance":"/usuarios/99","status":404,"title":"Not Found"}
```

**Por que assim:**

- O `ProblemDetail` é o formato padrão de erro do Spring desde a versão 6 e segue a RFC 9457: quem consome a API lê `status` e `detail` sempre no mesmo lugar, em qualquer erro.
- A resposta é um record separado da entidade: documento e e-mail, que são dados pessoais, não saem nesta rota, e a versão da trava otimista também não.
- O banco repete regras do Java (`unique`, `check (saldo >= 0)`). Um bug, um script manual ou outro serviço que grave na tabela não conseguem deixar um saldo negativo.
- O H2 em modo PostgreSQL deixa o projeto rodar sem Docker, mas não é o PostgreSQL. Tipos como `jsonb` e funções específicas não existem nele; por isso, a produção (e, idealmente, os testes de integração, com Testcontainers) usam o banco de verdade.
