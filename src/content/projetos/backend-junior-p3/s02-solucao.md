```java executar
import java.math.BigDecimal;
import java.util.HashMap;
import java.util.Map;

public class RegrasDaCarteira {
    static class RegraDeNegocioException extends RuntimeException {
        RegraDeNegocioException(String mensagem) {
            super(mensagem);
        }
    }

    enum Tipo { COMUM, LOJISTA }

    static class Usuario {
        final String nome;
        final Tipo tipo;
        BigDecimal saldo;

        Usuario(String nome, Tipo tipo, String saldo) {
            this.nome = nome;
            this.tipo = tipo;
            this.saldo = new BigDecimal(saldo);
        }

        void debitar(BigDecimal valor) {
            if (tipo == Tipo.LOJISTA) throw new RegraDeNegocioException("lojistas só recebem transferências");
            if (saldo.compareTo(valor) < 0) throw new RegraDeNegocioException("saldo insuficiente"); //> compareTo: com equals, 10.0 e 10.00 seriam diferentes.
            saldo = saldo.subtract(valor);
        }

        void creditar(BigDecimal valor) {
            saldo = saldo.add(valor);
        }
    }

    static final Map<Long, Usuario> usuarios = new HashMap<>(Map.of(
            1L, new Usuario("Ana Lima", Tipo.COMUM, "500.00"),
            2L, new Usuario("Bruno Reis", Tipo.COMUM, "50.00"),
            3L, new Usuario("Loja Ipê", Tipo.LOJISTA, "0.00")));

    static String transferir(String valor, long de, long para) {
        try {
            if (de == para) throw new RegraDeNegocioException("pagador e recebedor precisam ser diferentes");
            Usuario pagador = usuarios.get(de), recebedor = usuarios.get(para);
            BigDecimal v = new BigDecimal(valor);
            pagador.debitar(v);
            recebedor.creditar(v);
            return "ok";
        } catch (RegraDeNegocioException e) {
            return "422 " + e.getMessage();
        }
    }

    public static void main(String[] args) {
        System.out.println("1 -> 2, 100.00: " + transferir("100.00", 1, 2));
        System.out.println("2 -> 3, 500.00: " + transferir("500.00", 2, 3));
        System.out.println("3 -> 1, 10.00:  " + transferir("10.00", 3, 1));
        System.out.println("1 -> 1, 10.00:  " + transferir("10.00", 1, 1));
        System.out.println("2 -> 1, 150.00: " + transferir("150.00", 2, 1));
        usuarios.entrySet().stream().sorted(Map.Entry.comparingByKey())
                .forEach(e -> System.out.println(e.getKey() + " " + e.getValue().nome + ": " + e.getValue().saldo));
    }
}
```

**`TransferenciaService.java`**

```java
package br.com.ipe.carteira;

import java.time.Clock;
import java.time.OffsetDateTime;
import java.time.temporal.ChronoUnit;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class TransferenciaService {

    private final UsuarioRepository usuarios;
    private final TransferenciaRepository transferencias;
    private final Clock relogio;

    public TransferenciaService(UsuarioRepository usuarios, TransferenciaRepository transferencias, Clock relogio) {
        this.usuarios = usuarios;
        this.transferencias = transferencias;
        this.relogio = relogio;
    }

    @Transactional
    public TransferenciaResposta transferir(NovaTransferencia pedido) { //> Uma RuntimeException aqui desfaz o débito e o crédito.
        if (pedido.pagador().equals(pedido.recebedor())) {
            throw new RegraDeNegocioException("pagador e recebedor precisam ser diferentes");
        }
        Usuario pagador = buscar(pedido.pagador());
        Usuario recebedor = buscar(pedido.recebedor());
        pagador.debitar(pedido.valor());
        recebedor.creditar(pedido.valor());
        Transferencia t = new Transferencia(pagador, recebedor, pedido.valor(),
                OffsetDateTime.now(relogio).truncatedTo(ChronoUnit.MICROS)); //> O banco guarda microssegundos; a resposta precisa bater com ele.
        return TransferenciaResposta.de(transferencias.save(t));
    }

    @Transactional(readOnly = true)
    public TransferenciaResposta buscarTransferencia(Long id) {
        return transferencias.findById(id)
                .map(TransferenciaResposta::de)
                .orElseThrow(() -> new NaoEncontradoException("transferência " + id + " não existe"));
    }

    private Usuario buscar(Long id) {
        return usuarios.findById(id)
                .orElseThrow(() -> new NaoEncontradoException("usuário " + id + " não existe"));
    }
}
```

**`TratadorDeErros.java`**

```java
package br.com.ipe.carteira;

import java.util.stream.Collectors;
import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

@RestControllerAdvice
public class TratadorDeErros {

    @ExceptionHandler(NaoEncontradoException.class)
    public ProblemDetail naoEncontrado(NaoEncontradoException e) {
        return ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, e.getMessage());
    }

    @ExceptionHandler(RegraDeNegocioException.class)
    public ProblemDetail regra(RegraDeNegocioException e) {
        return ProblemDetail.forStatusAndDetail(HttpStatus.UNPROCESSABLE_CONTENT, e.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ProblemDetail invalida(MethodArgumentNotValidException e) {
        String campos = e.getBindingResult().getFieldErrors().stream()
                .map(f -> f.getField() + ": " + f.getDefaultMessage())
                .sorted()
                .collect(Collectors.joining("; "));
        return ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST, campos);
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ProblemDetail ilegivel(HttpMessageNotReadableException e) {
        return ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST, "corpo da requisição inválido");
    }
}
```

**Por que assim:**

- O `debitar` confere lojista e saldo antes de mexer em qualquer coisa: se a regra falha, o objeto fica como estava. É a mesma ideia do "tudo ou nada", no nível de um objeto.
- O horário é truncado para microssegundos porque é a precisão do `timestamp` do banco. Sem isso, a resposta do `POST` mostrava `.2403629` e a consulta depois mostrava `.240363`: o mesmo registro com dois valores diferentes.
- As chaves estrangeiras ganharam índices na migration. O PostgreSQL não cria índice para chave estrangeira sozinho, e as consultas "transferências de um usuário" vão precisar deles.
- Dois pagamentos simultâneos da mesma pessoa são protegidos pelo `@Version` da etapa 1: o segundo `update` não encontra a versão lida e falha. Essa falha ainda vira `500`; tratá-la como `409` é um bom exercício extra.
